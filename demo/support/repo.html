def _sanitize_no_proxy() -> None:
    """Drop bracketed IPv6 literals (e.g. ``[::1]``) from no_proxy/NO_PROXY.

    OpenMinion's providers use httpx with ``trust_env=True``, and httpx
    raises ``InvalidURL: Invalid port: ':1]'`` on bracketed IPv6 entries in
    no_proxy. Bare IPv6 forms (``::1``) parse fine, and localhost routing is
    unaffected. Only this process's environment is touched, so the
    local Ollama server is reached directly instead of through the proxy.
    """
    for var in ("no_proxy", "NO_PROXY"):
        value = os.environ.get(var)
        if not value:
            continue
        kept = [part for part in value.split(",") if not part.startswith("[")]
        for host in ("localhost", "127.0.0.1"):
            if host not in kept:
                kept.append(host)
        os.environ[var] = ",".join(kept)

"""Read-only research tools for the Research Brief Agent.

All tools are pure reads over public HTTP endpoints and need no API key:
Wikipedia's public API for search/summaries, plus a generic URL fetcher.
They are declared READ_ONLY + idempotent so OpenMinion's default action
policy runs them without an interactive confirmation.

Nothing here calls a model; the model is OpenMinion's configured provider.
"""

from __future__ import annotations

import html
import ipaddress
import json
import re
import urllib.error
import urllib.parse
import urllib.request
from socket import timeout as SocketTimeout

from openminion import tool

_USER_AGENT = "ResearchBriefAgent/0.1 (Cortensor Hackathon #5; +https://github.com/anzdex)"
_MAX_CHARS = 4000
_HTTP_TIMEOUT_S = 15


def _get_json(url: str) -> dict:
    req = urllib.request.Request(url, headers={"User-Agent": _USER_AGENT})
    with urllib.request.urlopen(req, timeout=_HTTP_TIMEOUT_S) as resp:
        return json.loads(resp.read().decode("utf-8", "replace"))


def _get_text(url: str) -> str:
    req = urllib.request.Request(url, headers={"User-Agent": _USER_AGENT})
    with urllib.request.urlopen(req, timeout=_HTTP_TIMEOUT_S) as resp:
        raw = resp.read().decode("utf-8", "replace")
    # Strip scripts/styles, then tags; keep readable text only.
    raw = re.sub(r"(?is)<(script|style)[^>]*>.*?</\1>", " ", raw)
    raw = re.sub(r"(?s)<[^>]+>", " ", raw)
    text = html.unescape(re.sub(r"\s+", " ", raw)).strip()
    return text[:_MAX_CHARS]


def _is_public_host(host: str) -> bool:
    """Block localhost / private-network fetches (SSRF guard)."""
    host = host.lower().rstrip(".")
    if host in {"localhost", "ip6-localhost"}:
        return False
    try:
        return not ipaddress.ip_address(host).is_private
    except ValueError:
        return True  # a DNS name: resolve check happens at connect time
